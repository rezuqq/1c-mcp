&НаКлиенте
Процедура ВыгрузитьКартинкуДляНоменклатуры(СсылкаНоменклатуры, КодНоменклатуры, КаталогКартинок)
    
    // Получаем адрес картинки из реквизита ФайлКартинки справочника Номенклатура
    АдресКартинки = ПолучитьАдресКартинки(СсылкаНоменклатуры, ЭтаФорма.УникальныйИдентификатор);
    
    Если АдресКартинки <> Неопределено И АдресКартинки <> "" Тогда
        
        // Получаем двоичные данные по адресу
        ДвоичныеДанные = ПолучитьИзВременногоХранилища(АдресКартинки);
        
        Если ДвоичныеДанные <> Неопределено Тогда
            // Формируем имя файла (используем jpg как расширение по умолчанию)
            ИмяФайла = СокрЛП(КодНоменклатуры) + "_01.jpg";
            ПолныйПуть = КаталогКартинок + ИмяФайла;
            
            // Сохраняем файл на диск
            ДвоичныеДанные.Записать(ПолныйПуть);
        КонецЕсли;
    КонецЕсли;
    
КонецПроцедуры

// Получение списка номенклатуры, у которой заполнен реквизит ФайлКартинки
&НаСервере
Функция ПолучитьНоменклатуруСКартинкамиНаСервере(МассивСсылокНаТовар)
    
    Запрос = Новый Запрос;
    Запрос.Текст = 
    "ВЫБРАТЬ
    |    Номенклатура.Ссылка КАК Ссылка,
    |    Номенклатура.Код КАК Код
    |ИЗ
    |    Справочник.Номенклатура КАК Номенклатура
    |ГДЕ
    |    Номенклатура.Ссылка В (&МассивСсылокНаТовар)
    |    И НЕ Номенклатура.ЭтоГруппа
    |    И Номенклатура.ФайлКартинки <> НЕОПРЕДЕЛЕНО
    |    И НЕ Номенклатура.ПометкаУдаления
    |
    |УПОРЯДОЧИТЬ ПО
    |    Номенклатура.Код";
    
    Запрос.УстановитьПараметр("МассивСсылокНаТовар", МассивСсылокНаТовар);
    
    Результат = Запрос.Выполнить();
    
    МассивНоменклатуры = Новый Массив;
    
    Выборка = Результат.Выбрать();
    Пока Выборка.Следующий() Цикл
        СтруктураДанных = Новый Структура;
        СтруктураДанных.Вставить("Ссылка", Выборка.Ссылка);
        СтруктураДанных.Вставить("Код", Выборка.Код);
        МассивНоменклатуры.Добавить(СтруктураДанных);
    КонецЦикла;
    
    Возврат МассивНоменклатуры;
    
КонецФункции

    ЗапросСвойств.УстановитьПараметр("СвойствоЗапретитьВсеСкидки", ПланыВидовХарактеристик.ДополнительныеРеквизитыИСведения.НайтиПоНаименованию("ЗапретитьВсеСкидки"));
	
	
    РезультатСвойств = ЗапросСвойств.Выполнить();
    
    // 3. Кэширование свойств
    СоответствиеСвойств = Новый Соответствие;
    ВыборкаСвойств = РезультатСвойств.Выбрать();
    Пока ВыборкаСвойств.Следующий() Цикл
        Ключ = ВыборкаСвойств.Номенклатура;
        Если НЕ СоответствиеСвойств[Ключ] = Неопределено Тогда
            СписокСвойств = СоответствиеСвойств[Ключ];
        Иначе
            СписокСвойств = Новый Массив;
            СоответствиеСвойств.Вставить(Ключ, СписокСвойств);
        КонецЕсли;
        
        СтруктураСвойства = Новый Структура;
        СтруктураСвойства.Вставить("Имя", Строка(ВыборкаСвойств.Свойство));
        СтруктураСвойства.Вставить("Значение", Строка(ВыборкаСвойств.Значение));
        СписокСвойств.Добавить(СтруктураСвойства);
    КонецЦикла;
    
    // 4. Сбор уникальных имен свойств для заголовков
    УникальныеСвойства = Новый Массив;
    ВременноеСоответствие = Новый Соответствие;
    
    Для каждого Ключ Из СоответствиеСвойств Цикл
        МассивСвойств = Ключ.Значение;
        Для каждого СтрокаСвойства Из МассивСвойств Цикл
            Если ВременноеСоответствие[СтрокаСвойства.Имя] = Неопределено Тогда
                ВременноеСоответствие.Вставить(СтрокаСвойства.Имя, Истина);
                УникальныеСвойства.Добавить(СтрокаСвойства.Имя);
            КонецЕсли;
        КонецЦикла;
    КонецЦикла;
    
    // 5. Формирование CSV текста в памяти
    МассивСтрок = Новый Массив;
    
    // Формируем заголовок 
	
	
    Заголовки = Новый Массив;
    Заголовки.Добавить("Код");
    Заголовки.Добавить("Наименование");
    Заголовки.Добавить("Родитель");
    Заголовки.Добавить("Артикул");
    Заголовки.Добавить("Цена");
    Заголовки.Добавить("ЭтоГруппа");
    Заголовки.Добавить("Остаток");
    Заголовки.Добавить("Описание");
    Заголовки.Добавить("Валюта");
    
    Для каждого ИмяСвойства Из УникальныеСвойства Цикл
        Заголовки.Добавить(ИмяСвойства);
    КонецЦикла;
    
    МассивСтрок.Добавить(СтрСоединить(Заголовки, Объект.СимволРазделитель));
    
    // 6. Формирование строк данных
    Выборка = РезультатЗапроса.Выбрать(ОбходРезультатаЗапроса.ПоГруппировкам);
    
    Пока Выборка.Следующий() Цикл
        ЗначенияСтроки = Новый Массив;
        
        ЗначенияСтроки.Добавить(Выборка.Код);
        // Основные поля        
        мНаименование = СтрЗаменить(Выборка.Наименование, Символы.ПС,"");
        мНаименование = СтрЗаменить(мНаименование, Символы.ВК,"");
        ЗначенияСтроки.Добавить(мНаименование);
        ЗначенияСтроки.Добавить(""+Выборка.Родитель);
        ЗначенияСтроки.Добавить(Выборка.Артикул);
        ЗначенияСтроки.Добавить(Формат(Выборка.Цена,"ЧГ=0"));
        ЗначенияСтроки.Добавить(Строка(Выборка.ЭтоГруппа));
        ЗначенияСтроки.Добавить(Формат(Выборка.Остаток,"ЧГ=0"));
        ЗначенияСтроки.Добавить(Выборка.Описание);
        ЗначенияСтроки.Добавить(Выборка.ВалютаЦены);
        
        // Дополнительные свойства
        Если НЕ Выборка.ЭтоГруппа Тогда
            МассивСвойствЭлемента = СоответствиеСвойств[Выборка.Ссылка];
            
            Если МассивСвойствЭлемента <> Неопределено Тогда
                // Создаем соответствие для быстрого поиска
                ВремСвойства = Новый Соответствие;
                Для каждого СтрокаСвойства Из МассивСвойствЭлемента Цикл
                    ВремСвойства.Вставить(СтрокаСвойства.Имя, СтрокаСвойства.Значение);
                КонецЦикла;
                
                Для каждого ИмяСвойства Из УникальныеСвойства Цикл
                    Если ВремСвойства[ИмяСвойства] <> Неопределено Тогда
                        ЗначенияСтроки.Добавить(ВремСвойства[ИмяСвойства]);
                    Иначе
                        ЗначенияСтроки.Добавить("");
                    КонецЕсли;
                КонецЦикла;
            Иначе
                // Нет свойств - добавляем пустые значения
                Для Сч = 1 По УникальныеСвойства.Количество() Цикл
                    ЗначенияСтроки.Добавить("");
                КонецЦикла;
            КонецЕсли;
        Иначе
            // Для групп оставляем пустые значения свойств
            Для Сч = 1 По УникальныеСвойства.Количество() Цикл
                ЗначенияСтроки.Добавить("");
            КонецЦикла;
        КонецЕсли;
        
        МассивСтрок.Добавить(СтрСоединить(ЗначенияСтроки, Объект.СимволРазделитель));
    КонецЦикла;
    
    // 7. Объединяем все строки
    ТекстCSV = СтрСоединить(МассивСтрок, Символы.ПС);
    
    Возврат ТекстCSV;
    
КонецФункции
