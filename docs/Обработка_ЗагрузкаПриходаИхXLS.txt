Функция ПоискТовараПоКодуПОставщика(пПартнер, пКодНоменклПоставщика)	   

	мВозвр = Новый Структура;
	мВозвр.Вставить("Номенклатура1С",Справочники.Номенклатура.ПустаяСсылка());
	мВозвр.Вставить("НоменклПоставщикаИз1С",Справочники.НоменклатураКонтрагентов.ПустаяСсылка());
	
	
	мЗапрос = Новый Запрос;
	мЗапрос.Текст = "
		|выбрать первые 1
		|	СпрКодыПоставщиков.*
		|ИЗ
		|	Справочник.НоменклатураКонтрагентов как СпрКодыПоставщиков 
		|Где
		|	Не СпрКодыПоставщиков.ПометкаУдаления
		|	И СпрКодыПоставщиков.Владелец = &Партнер	
		|	И ((СпрКодыПоставщиков.Код = &КодНоменклПоставщика) ИЛИ (СпрКодыПоставщиков.КодНоменклатуры = &КодНоменклПоставщика))
		|";
	
	
	
	мЗапрос.УстановитьПараметр("Партнер", пПартнер);	
	мЗапрос.УстановитьПараметр("КодНоменклПоставщика", СокрЛП(пКодНоменклПоставщика));	 
	
	мРезТаб = мЗапрос.Выполнить().Выгрузить();
	
	Если мРезТаб.Количество() > 0 Тогда
		мВозвр.Вставить("Номенклатура1С", мРезТаб[0].Ссылка.Номенклатура);
		мВозвр.Вставить("НоменклПоставщикаИз1С", мРезТаб[0].Ссылка);
	КонецЕсли;	
	
	
	возврат мВозвр;


КонецФункции

		|Где

		|	И СпрН.Наименование ПОДОБНО &КодНоменклПоставщика

Функция ПолучитьГТДИзСправочника(пНомерГТД, пСтрана)

	мЗапрос = Новый Запрос();
	мЗапрос.Текст = "
		|ВЫБРАТЬ Первые 1
		|	НомераГТД.Ссылка КАК Ссылка
		|ИЗ
		|	Справочник.НомераГТД КАК НомераГТД
		|ГДЕ
		|	НомераГТД.Код = &Код";   
	
	мЗапрос.УстановитьПараметр("Код", пНомерГТД);
	
	
	мВыборка = мЗапрос.Выполнить().Выбрать();
	Если мВыборка.Следующий() Тогда
		Возврат мВыборка.Ссылка;

Функция ПолучитьСтавкуНДСИзЧисла(СтавкаЧисло)
    // Переменная для хранения числового значения ставки НДС
    СтавкаНДСИзПеречисления = Справочники.СтавкиНДС.БезНДС;
    // Проверяем, какая ставка НДС передана и присваиваем соответствующее значение
	Если СтавкаЧисло = 20  Тогда 
		СтавкаНДСИзПеречисления = Перечисления.СтавкиНДС.НДС20;
	ИначеЕсли СтавкаЧисло = 10  Тогда 
		СтавкаНДСИзПеречисления = Перечисления.СтавкиНДС.НДС10;
	ИначеЕсли СтавкаЧисло = 7  Тогда 
		СтавкаНДСИзПеречисления = Перечисления.СтавкиНДС.НДС7;
	ИначеЕсли СтавкаЧисло = 5  Тогда 
		СтавкаНДСИзПеречисления = Перечисления.СтавкиНДС.НДС5;
	Иначе
		СтавкаНДСИзПеречисления = Перечисления.СтавкиНДС.БезНДС;
	КонецЕсли;     
	
	СтавкаНДСИзПеречисления = Справочники.СтавкиНДС.НайтиПоРеквизиту("ПеречислениеСтавкаНДС",СтавкаНДСИзПеречисления);

	Возврат СтавкаНДСИзПеречисления;
КонецФункции
